// Football-Data.org API Integration const API_KEY = "YOUR_FOOTBALL_DATA_API_KEY"; // Replace with your key from football-data.org const BARCA_ID = 81; async function fetchBarcaData() { const headers = { "X-Auth-Token": API_KEY }; try { // 1. Fetch FC Barcelona Matches const matchRes = await fetch(`https://api.football-data.org/v4/teams/${BARCA_ID}/matches?limit=10`, { headers }); const matchData = await matchRes.json(); if (matchData.matches) { renderMatchInfo(matchData.matches); } } catch (err) { console.error("Error connecting to Football-Data.org:", err); } } function renderMatchInfo(matches) { // Extract Last Finished Match const lastMatch = matches.filter(m => m.status === 'FINISHED').pop(); if (lastMatch) { const isHome = lastMatch.homeTeam.id === BARCA_ID; const opponent = isHome ? lastMatch.awayTeam.name : lastMatch.homeTeam.name; const score = `${lastMatch.score.fullTime.home} - ${lastMatch.score.fullTime.away}`; document.getElementById('last-match-title').textContent = `${isHome ? 'Barça' : opponent} ${score} ${isHome ? opponent : 'Barça'}`; } // Extract Next Upcoming Match const nextMatch = matches.find(m => m.status === 'SCHEDULED' || m.status === 'TIMED'); if (nextMatch) { const isHome = nextMatch.homeTeam.id === BARCA_ID; const opponent = isHome ? nextMatch.awayTeam.name : nextMatch.homeTeam.name; const matchTime = new Date(nextMatch.utcDate).toLocaleString(); document.getElementById('next-match-title').textContent = `vs ${opponent}`; document.getElementById('next-match-time').textContent = matchTime; } } document.addEventListener('DOMContentLoaded', fetchBarcaData);
